; sega2asm symbols — International Superstar Soccer Deluxe (Europe)
; SHA-1: CCC60352B43F8C3D536267DD05A8F2C0F3B73DF6
;
; Format: name = $ADDRESS ; comment
; The comment is written above the label in the generated assembly
; (use \n for line breaks). RAM variables are emitted as equates in
; asm/include/variables.asm and used symbolically, e.g. (g_next_state-$FF0000)(a6).
;
; Naming: functions are snake_case, RAM globals g_*, ROM data tables tbl_* /
; dat_*. Addresses without a curated name keep their sub_/loc_/dat_ names.

; ═════════════════════════════════════════════════════════════════════════════
; Reset / boot
; ═════════════════════════════════════════════════════════════════════════════
reset                       = $000200 ; Reset vector: standard Sega TMSS / VDP / Z80 initialisation (ICD_BLK4)
reset_init_table            = $00028E ; Register values, VDP register list and Z80 stub used by reset
vblank_irq                  = $000312 ; Level 6 interrupt: calls the handler stored in g_vblank_handler
hblank_irq                  = $000320 ; Level 4 interrupt: calls the handler stored in g_hblank_handler
install_irq_vectors_unused  = $00032E ; Unreferenced: writes the IRQ vectors into low memory and unlocks TMSS
main_init                   = $000358 ; Cold start: runs the four boot screens (Konami, Factor 5, ...) then warm starts
warm_start                  = $0003B8 ; Re-entry point: boot module screen 1 decides between the game and the demo path
main_loop_dispatch          = $000460 ; Jump to g_next_state (also remembered as g_current_state).\nEvery game state ends by setting g_next_state and jumping here.

; ═════════════════════════════════════════════════════════════════════════════
; Sprite attribute buffer (64 hardware sprites, DMA'd to VRAM $1C00)
; ═════════════════════════════════════════════════════════════════════════════
sprites_init                = $00046A ; Clear the sprite buffer (link fields 1..64) and queue its DMA to VRAM $1C00
sprites_end_frame           = $0004B4 ; Hide unused sprite slots, rewind the buffer and queue a priority DMA to VRAM $1C00
sprite_alloc                = $0004F8 ; Return a1 = next free sprite slot (d1 = -1 when all 64 are used)

; ═════════════════════════════════════════════════════════════════════════════
; Palette fades (implemented as objects; target palette g_palette_target,
; working copy g_palette_current, step counter g_fade_step)
; ═════════════════════════════════════════════════════════════════════════════
fade_out_start              = $000514 ; Start a fade to black (object) if no fade is running; d0 = -1 when busy
fade_out_step               = $000568 ; Fade object callback: darken g_palette_current one step per frame
fade_in_start               = $0005DA ; Start a fade from black to g_palette_target
fade_in_step                = $000618 ; Fade object callback: brighten towards g_palette_target
fade_white_start            = $0006E4 ; Start a fade to white
fade_white_step             = $00071A ; Fade object callback: brighten every colour towards white
fade_white_back_step        = $000796 ; Fade object callback: from white back to g_palette_target

; ═════════════════════════════════════════════════════════════════════════════
; DMA queue (64 entries x 20 bytes at g_dma_queue, flushed during VBlank
; with a per-frame budget of $1392 words)
; ═════════════════════════════════════════════════════════════════════════════
dma_queue_init              = $000846 ; Reset the DMA queue read / write pointers
dma_queue_flush             = $00085C ; Execute queued DMA transfers until the frame budget is used up
dma_queue_add               = $00092C ; Queue a VRAM DMA: d0 = auto-increment, d1 = length (bytes), d2 = source,\nd3 = VRAM destination, d4 = cost. Splits transfers crossing a 128 KB boundary.
dma_queue_add_front         = $000B30 ; Like dma_queue_add but inserts in front of the queue (runs first)
dma_queue_add_cram          = $000BD0 ; Queue a CRAM DMA (palette upload): d1 = length, d2 = source, d3 = CRAM address

; ═════════════════════════════════════════════════════════════════════════════
; System initialisation and VDP
; ═════════════════════════════════════════════════════════════════════════════
system_init                 = $000C6E ; Clear work RAM, VRAM and CRAM, validate the saved settings checksum and\nreset the options / team tables to their defaults
settings_checksum           = $000D92 ; Recompute the checksum of the settings block at $FFFFF0 into $FFFFEE
vdp_init_game               = $000DAE ; Install the default IRQ handlers and program the VDP for the game:\nH32 (256 px), 64x32 planes, plane A $2000, plane B $0000, window $1000,\nsprites $1C00, hscroll $1800, shadow/highlight enabled
vblank_default              = $000E5C ; Default VBlank handler: tick the sound driver
hblank_default              = $000E94 ; Default HBlank handler
vdp_write_scroll            = $000E9E ; Write the plane A/B horizontal and vertical scroll values, then wait so the\nframe takes the same time on 50 and 60 Hz machines
debug_color_red             = $000F0A ; Debug: set CRAM colour $20 (profiling bar colours)
decompress_factor5          = $000F82 ; Unpack a Factor 5 "P1"/"P2"/"P3" stream: a0 = source, a1 = destination.\nOn return a1 points past the unpacked data.
joypad_read_all             = $00102A ; Read both controller ports (with multitap support) into g_pad_type / g_pad_state
joypad_read_port            = $00106C ; Identify and read the device on the port at a0 (3/6-button pad, multitap)
joypad_th_toggle            = $0011C4 ; Toggle the TH line of the port at a0 and read the data register
joypad_th_handshake         = $0011F2 ; TH/TR handshake used by the multitap protocol (d7 = timeout)

; ═════════════════════════════════════════════════════════════════════════════
; Maths and steering. Angles are 0-63: 0 = up the pitch (-Y), 16 = right (+X),
; 32 = down, 48 = left. Speeds are 16.16 pixels per frame and come in
; (NTSC, PAL) pairs, the PAL value being 1.2x the NTSC one.
; ═════════════════════════════════════════════════════════════════════════════
tbl_speed_ntsc_pal          = $00BBEE ; (NTSC, PAL) pairs of 16.16 speeds, indexed by g_is_pal
direction_to                = $00BDAC ; d0 = angle 0-63 from object a5 to point (d0, d1): table-driven atan2
tbl_atan_octants            = $00BEAE ; Angle results for direction_to, one block per quadrant
tbl_direction_x             = $00BFCA ; 64 signed 8.8 X components of the unit vector for each angle;\nthe Y components follow 64 entries later (shaped for the oblique pitch)
obj_steer_idle              = $010D7C ; Think callback: stand still, remember the current position as the target
obj_steer_to_target         = $010D98 ; Think callback: turn towards obj_target_x/y until within 16 pixels
obj_steer_ahead             = $010E02 ; Think callback: project the target obj_travel pixels ahead along obj_facing

; ═════════════════════════════════════════════════════════════════════════════
; Object system (doubly linked lists; +0 next, +4 prev, +$34 update callback)
; ═════════════════════════════════════════════════════════════════════════════
obj_alloc                   = $00C124 ; Take an object from the free list into the active list (a5 = object)
obj_link_active             = $00C156 ; Insert object a5 at the head of the active list
obj_free                    = $00C174 ; Unlink object a5 from the active list and return it to the free list
objects_update              = $00C0CA ; Run every active object's think (+$38) and update (+$34) callbacks.\nWhile a fade is running only objects without a draw callback are updated.
objects_draw                = $00C1DC ; Depth-sort the active list by +$14 (pitch Y) and call each draw callback (+$66).\nScreen position: x' = x + y/2 - hscroll, y' = y/2 - z - vscroll (oblique pitch projection).
objects_draw_menu           = $00C2A4 ; Same as objects_draw for front-end screens

; ═════════════════════════════════════════════════════════════════════════════
; Game states. Each state initialises the hardware and its resources, installs
; a VBlank handler that runs the whole frame (the game logic runs inside the
; VBlank interrupt), fades in, then waits until g_frame_state returns to 0
; (fade-out finished) and jumps to g_next_state.
; ═════════════════════════════════════════════════════════════════════════════
state_menu                  = $01FF16 ; Front-end state: menu screens are built from objects
state_menu_frame            = $01FF6E ; VBlank handler of state_menu
state_screen                = $01FFF4 ; Full-screen presentation state (loads its screen with sub_01E294)
state_screen_frame          = $02004C ; VBlank handler of state_screen
state_match                 = $0200D2 ; Match state: loads stadium, teams and players, then plays the game
state_match_frame           = $02014E ; VBlank handler of state_match: scroll, DMA, sound, pads, match logic, objects, sprites
state_result                = $0201FE ; Post-match / presentation state
state_result_frame          = $020274 ; VBlank handler of state_result
player_init_sprite          = $020BD4 ; Give player object a5 its draw routine and an empty animation state
player_draw                 = $020BF0 ; Player draw callback: pick the frame from tbl_player_anims[action $6A][direction][frame $7A],\nDMA its body/head/hair/kit tiles into the player's VRAM slot ($78) and emit its sprite pieces
tbl_player_anims            = $0210DE ; 54 actions -> 8 directions -> frame descriptor lists.\nDescriptor: +0 res00 entry*4, +2 tile offset, +4 tile bytes, +6 head, +8 kit, +A hair tile offsets,\n+C piece count-1, then 10-byte pieces: y, VDP size, attribute index, tile, x (right), x (left)
tbl_sprite_attr_right       = $020F5E ; Sprite attribute words (palette line / flips) indexed by piece attribute byte, right-facing
tbl_sprite_attr_left        = $02101E ; Same with horizontal flip toggled, used for left-facing directions
tbl_player_names            = $035916 ; 43 pointers (one per team) to 20 x 8-character player names ('@' = space)
tbl_player_data             = $038140 ; 43 pointers to 20 x 12-byte player records, copied to player object +$5A..+$65\n(8 attribute bytes, body type, face style, hair style, extra)
tbl_team_ratings            = $03AB82 ; 5 rating bytes per team (copied to $183C / $18C4)
tbl_kit_clash               = $0374AE ; Word per team: when home and away values match, the away team wears its second kit
tbl_head_mask_home          = $037402 ; Per-team AND masks applied to head tiles (skin / hair colours), home team
tbl_head_mask_away          = $037458 ; Same for the away team
load_stadium_tiles          = $01D7BA ; Load stadium group 7 + g_stadium: tiles (entries 3, 4, night patch 5), metatile map (1)\nand table (2) into RAM, palette line 3 for g_time_of_day (6-8); create the corner-flag objects
camera_update               = $01DBBE ; Per-frame camera: ease towards g_camera_focus, clamp to the pitch and stream\nnew metatile rows / columns into plane B
pitch_draw_row              = $01DD88 ; Draw pitch cell row d7 starting at cell column d6 into plane B (via the DMA queue)
pitch_draw_column           = $01DE54 ; Draw pitch cell column d6 starting at cell row d7 into plane B

; ═════════════════════════════════════════════════════════════════════════════
; Front-end text and tilemap helpers. They draw into the off-screen nametable
; pointed to by $1762(a6) (64 cells wide, $80 bytes per row); coordinates are
; given in pixels. Glyph tile = character - '-' ($2D) + font base tile.
; ═════════════════════════════════════════════════════════════════════════════
rect_highlight              = $01F076 ; Set the palette bit ($2000) of every cell in the rectangle d0..d1 x d2..d3
rect_unhighlight            = $01F0DC ; Clear the palette bit of every cell in the rectangle
text_draw_large             = $01F29E ; Draw the $FF-terminated string at a0 at (d0,d1) with the 8x16 font
text_draw_small             = $01F358 ; Draw the $FF-terminated string at a0 at (d0,d1) with the 8x8 font
rect_fill_tiles             = $01F38A ; Fill the rectangle d0..d1 x d2..d3 with consecutive tiles starting at d4 (column-major)

; Modules

boot_main                   = $144318 ; Boot module entry: d0 = screen number (0-4). Runs one boot/logo screen.

sound_api_nop_3             = $1FD978 ; No operation

sound_api_nop_5             = $1FD980 ; No operation
